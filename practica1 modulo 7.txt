PRACTICA 1 - NFS (Modulo VII)
Servidor: Red Hat 10.0.0.36
Cliente:  Red Hat 10.0.0.37
(Enciende el servidor ANTES que el cliente)

=====================================================
0. SERVIDOR (10.0.0.36) y CLIENTE (10.0.0.37) - que no se apaguen (opcional)
=====================================================
sudo systemctl mask sleep.target suspend.target hibernate.target hybrid-sleep.target
gsettings set org.gnome.desktop.session idle-delay 0
gsettings set org.gnome.desktop.screensaver lock-enabled false

=====================================================
1. SERVIDOR (10.0.0.36) - instalar NFS y crear directorio con 100 archivos
=====================================================
sudo dnf install -y nfs-utils
sudo mkdir -p /OS3
cd /OS3 && sudo touch Adrian{1..100}.txt
ls /OS3 | wc -l

=====================================================
2. SERVIDOR (10.0.0.36) - permisos
=====================================================
sudo chmod -R 777 /OS3
sudo chown -R nobody:nobody /OS3
ls -ld /OS3

=====================================================
3. SERVIDOR (10.0.0.36) - exports
=====================================================
echo "/OS3 10.0.0.0/24(rw,sync,no_subtree_check)" | sudo tee -a /etc/exports
sudo exportfs -rav

=====================================================
4. SERVIDOR (10.0.0.36) - servicio y firewall
=====================================================
sudo systemctl enable --now nfs-server
sudo firewall-cmd --permanent --add-service={nfs,mountd,rpc-bind}
sudo firewall-cmd --reload
sudo exportfs -v

=====================================================
5. CLIENTE (10.0.0.37) - instalar cliente y montar a mano
=====================================================
sudo dnf install -y nfs-utils
sudo mkdir -p /mnt/OS3
showmount -e 10.0.0.36
sudo mount -t nfs 10.0.0.36:/OS3 /mnt/OS3
ls /mnt/OS3
ls /mnt/OS3 | wc -l
sudo umount /mnt/OS3

=====================================================
6. CLIENTE (10.0.0.37) - fstab (montaje automatico)
=====================================================
echo "10.0.0.36:/OS3  /mnt/OS3  nfs  defaults,_netdev,x-systemd.automount,x-systemd.mount-timeout=30  0  0" | sudo tee -a /etc/fstab
sudo systemctl daemon-reload
sudo systemctl restart remote-fs.target
cat /etc/fstab | grep OS3
ls /mnt/OS3 | wc -l
df -h | grep OS3

=====================================================
7. CLIENTE (10.0.0.37) - prueba final con reinicio
=====================================================
sudo reboot

# Al volver (primero ls, luego df):
ls /mnt/OS3 | wc -l
df -h | grep OS3

=====================================================
SI ALGO FALLA - CLIENTE (10.0.0.37)
=====================================================
sudo journalctl -b | grep -i nfs | tail
